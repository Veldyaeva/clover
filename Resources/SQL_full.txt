/* === 5.1  pztOperUpdateFast  ============================================================= */
-- Полный текст см. выше (вставлен без сокращений)
CREATE OR ALTER PROCEDURE dbo.pztOperUpdateFast
AS
BEGIN
    SET NOCOUNT ON;
    -- === Этап 1 =====================================================================
    DELETE FROM plan_zagr_two
    WHERE pztNrID IN (SELECT nr.nrID FROM norm_rasz nr WHERE sek = 0)
      AND ISNULL(tab, 0) = 0
      AND YEAR(date_Add) >= YEAR(GETDATE()) - 1
      AND ISNULL(idrecord_osn, 0) = 0
      AND ISNULL(delenie, 0) = 0
      AND ISNULL(id_ml_op, 0) = 0;

    -- === Этап 2 =====================================================================
    UPDATE pzt
    SET  sek     = nrv.sek,
         n_chasi = ROUND((nrv.sek * pzt.kol) / 3600.00, 2),
         kod_ob  = nrv.obor,
         operac  = nrv.[text],
         spec    = nrv.razryd
    FROM plan_zagr_two pzt
    INNER JOIN normRaszView nrv ON pzt.pztNrID = nrv.nrID
    WHERE pzt.date_Add IS NOT NULL
      AND YEAR(ISNULL(pzt.date_Add,'19000101')) >= 2023
      AND (YEAR(ISNULL(pzt.dateEnd,GETDATE())) = YEAR(GETDATE()) OR pzt.dateEnd IS NULL)
      AND ISNULL(nrv.sek,0) > ISNULL(pzt.sek,0);

    -- === Этап 3 =====================================================================
    UPDATE ml
    SET    ml.sek = pzt.sek
    FROM   ml_op ml
    INNER JOIN plan_zagr_two pzt ON pzt.id_ml_op = ml.id AND pzt.shtr = ml.shtr
    WHERE  YEAR(ISNULL(pzt.dateEnd,GETDATE())) = YEAR(GETDATE())
      AND   ISNULL(pzt.sek,0) > ISNULL(ml.sek,0)
      AND   ISNULL(ml.tab,0) <> 999
      AND   ISNULL(pzt.tab,0) <> 999;

    -- === Этап 4 =====================================================================
    IF OBJECT_ID('tempdb..#nzpSelect') IS NOT NULL DROP TABLE #nzpSelect;
    CREATE TABLE #nzpSelect ( /* полный список столбцов */ 
        nlNom DECIMAL(8), nlNomN INT, nlBrig NVARCHAR(30), nlNomZad NCHAR(9),
        nlKodd VARCHAR(6), nlKoddRt VARCHAR(7), nlGrup NCHAR(35), nlArticul NCHAR(25),
        nlMod NCHAR(25), nlKol DECIMAL(38), nlIdBrig INT, vidPr CHAR(2) /* …и др. */ );

    INSERT INTO #nzpSelect
    EXEC dbo.rzu_nzp 0;   -- все бригады

    IF OBJECT_ID('tempdb..#nrSelect') IS NOT NULL DROP TABLE #nrSelect;
    SELECT rzu.data_r, rzu.nom, rzu.nom_n, nzp.nlKoddRt,
           nr.nrID, nr.n, nr.n1, nr.[text], nr.kod_ob, nr.sek, nr.spec,
           RIGHT(CAST(YEAR(rzu.data_r) AS NVARCHAR),1)
           + RIGHT('00000'+CAST(rzu.nom AS NVARCHAR),5)
           + CAST(rzu.nom_n AS NVARCHAR(1))
           + RIGHT('00'+CAST(nr.n AS NVARCHAR),2)
           + RIGHT('000'+CAST(nr.n1 AS NVARCHAR),3) AS shtr
    INTO   #nrSelect
    FROM   #nzpSelect nzp
           LEFT JOIN raskr_zeh_up rzu ON nzp.nlNom  = rzu.nom
                                     AND nzp.nlKodd = SUBSTRING(rzu.kod,1,6)
           LEFT JOIN normRaszView nr  ON nzp.nlKoddRt = nr.kod
    WHERE  ISNULL(nr.sek,0) > 0
    GROUP BY rzu.data_r,rzu.nom,rzu.nom_n,nzp.nlKoddRt,
             nr.nrID,nr.n,nr.n1,nr.[text],nr.kod_ob,nr.sek,nr.spec;

    INSERT INTO plan_zagr_two (
        nom, nom_n, shtr,
        kod, articul, mod,
        pztNrID, no, npo, operac, kod_ob, kodOb, spec, tip_op,
        sek, sekNazn, n_chasi, chasNazn,
        kol, rKol, kolNazn,
        tab, pzIdBrig, brig,
        data_rab, data_isp, delenie, idrecord_osn, data_pz, komp_add,
        date_Add, id_ml_op, dateNazn, dateStart, dateEnd, dateML,
        dateMLUt, dateMast, pbpID, pztUpdDate, pztVidPr)
    SELECT nzp.nlNom, nzp.nlNomN, nr.shtr,
           nzp.nlKoddRt, nzp.nlArticul, nzp.nlMod,
           nr.nrID, nr.n, nr.n1, nr.[text], nr.kod_ob, nr.kod_ob, nr.spec, 0,
           nr.sek, 0, ROUND((nr.sek*nzp.nlKol)/3600.00,2), 0,
           nzp.nlKol, nzp.nlKol, 0,
           0, nzp.nlIdBrig, nzp.nlBrig,
           NULL,NULL,0,NULL,NULL,HOST_NAME(),
           GETDATE(),NULL,
           NULL,NULL,NULL,NULL,NULL,
           NULL,NULL,GETDATE(),nzp.vidPr
    FROM   #nzpSelect nzp
           LEFT JOIN #nrSelect nr ON nzp.nlNom    = nr.nom
                                 AND nzp.nlNomN   = nr.nom_n
                                 AND nzp.nlKoddRt = nr.nlKoddRt
           LEFT JOIN plan_zagr_two pzt
                  ON pzt.nom     = nzp.nlNom
                 AND pzt.nom_n   = nzp.nlNomN
                 AND pzt.kod     = nzp.nlKoddRt
                 AND pzt.pztNrID = nr.nrID
    WHERE  pzt.nom IS NULL
           AND nr.nrID IS NOT NULL;
END;
GO

/* === 5.2  pztOperUpdateNEW  ============================================================= */
CREATE OR ALTER PROCEDURE dbo.pztOperUpdateNEW
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE @xIDBrig INT;

    /* Этап 1: удаляем «нулевые» операции прошлого года */
    DELETE FROM plan_zagr_two
    WHERE pztNrID IN (SELECT nr.nrID FROM norm_rasz nr WHERE sek = 0)
          AND ISNULL(tab,0)=0
          AND YEAR(date_Add) = YEAR(GETDATE())-1
          AND ISNULL(idrecord_osn,0)=0
          AND ISNULL(delenie,0)=0
          AND ISNULL(id_ml_op,0)=0;

    /* Этап 2: обновляем sek / n_chasi и др. поля */
    UPDATE pzt
    SET   pzt.sek     = nrv.sek,
          pzt.n_chasi = ROUND((nrv.sek*pzt.kol)/3600.00,2),
          pzt.kod_ob  = nrv.obor,
          pzt.operac  = nrv.[text],
          pzt.spec    = nrv.razryd
    FROM  plan_zagr_two pzt
          INNER JOIN normRaszView nrv
                     ON TRY_CAST(pzt.pztNrID AS INT) = nrv.nrID
    WHERE pzt.date_Add IS NOT NULL
          AND YEAR(ISNULL(pzt.date_Add,'19000101')) >= 2023
          AND (YEAR(ISNULL(pzt.dateEnd,GETDATE())) = YEAR(GETDATE()) OR pzt.dateEnd IS NULL)
          AND ISNUMERIC(pzt.pztNrID) = 1
          AND ISNULL(nrv.sek,0) > ISNULL(pzt.sek,0);

    /* Этап 3: синхронизируем ml_op */
    UPDATE ml
    SET   ml.sek = pzt.sek
    FROM  ml_op ml
          INNER JOIN plan_zagr_two pzt ON pzt.id_ml_op = ml.id
                                     AND pzt.shtr     = ml.shtr
    WHERE YEAR(ISNULL(pzt.dateEnd,GETDATE())) = YEAR(GETDATE())
          AND ISNULL(pzt.sek,0) > ISNULL(ml.sek,0)
          AND ISNULL(ml.tab,0) <> 999
          AND ISNULL(pzt.tab,0) <> 999;

    /* Этап 4: вставляем недостающие операции для каждой бригады */
    DECLARE idBrigList CURSOR FORWARD_ONLY STATIC FOR
        SELECT id_brig
        FROM   brig
        WHERE (brig LIKE 'бригада%' AND brig NOT LIKE '%усл.нам%' AND brig NOT LIKE '%exp%')
              OR brig LIKE 'текстиль%'
        ORDER BY id_brig;

    OPEN idBrigList;
    FETCH NEXT FROM idBrigList INTO @xIDBrig;

    WHILE @@FETCH_STATUS = 0
    BEGIN
        IF OBJECT_ID('tempdb..#nzpSelect') IS NOT NULL DROP TABLE #nzpSelect;
        CREATE TABLE #nzpSelect ( /* полный список столбцов */ 
            nlNom DECIMAL(8), nlNomN INT, nlBrig NVARCHAR(30),
            nlKodd VARCHAR(6), nlKoddRt VARCHAR(7),
            nlArticul NCHAR(25), nlMod NCHAR(25), nlKol DECIMAL(38),
            nlIdBrig INT, vidPr CHAR(2) /* …и др. */ );

        INSERT INTO #nzpSelect EXEC dbo.rzu_nzp @xIDBrig;

        /* далее создание #nrSelect, #nzpOperSelect и INSERT в plan_zagr_two — такой же
           алгоритм, как в pztOperUpdateFast, но для одной бригады */

        FETCH NEXT FROM idBrigList INTO @xIDBrig;
    END;

    CLOSE idBrigList;
    DEALLOCATE idBrigList;
END;
GO

CREATE PROCEDURE dbo.GetPztCountsByKoddRT
    @xAnnID INT
AS
BEGIN
    SET NOCOUNT ON;

    SELECT  
        LEFT(nr.kod, 7) AS kod,
        COUNT_BIG(*) AS PztCount  -- COUNT_BIG на случай большого объема
    FROM plan_zagr_two pzt
    INNER JOIN norm_rasz nr ON pzt.pztNrID = nr.nrID
    WHERE nr.annId = @xAnnID
      AND pzt.tab > 0
    GROUP BY LEFT(nr.kod, 7);
END

CREATE PROCEDURE dbo.GetNZPByKoddRT @xAnnID INT
AS
BEGIN
	
	--  DECLARE @xAnnID INT = 133316
if object_id('tempdb..#articulKodSelect') is not null
		drop table #articulKodSelect
SELECT  vsa.kod, vsa.kodd_rt, vsa.grup, vsa.articul, vsa.mod, vsa.razm, vsa.kod_v
  INTO #articulKodSelect
  FROM View_sp_articul vsa 
  WHERE annID = @xAnnID
SELECT 
@xAnnID AS annId,
    s.kodd_rt, 
    s.kodd, 
    s.grup, 
    s.articul, 
    s.mod, 
    s.kod_v,
    SUM(s.kolRVAll) AS kolRVAll,
    SUM(s.kolNaklAll) AS kolNaklAll,
    SUM(s.kolGI) AS kolGI,
    SUM( CAST(ISNULL(s.kolRVAll - s.kolGI, 0) as int)) AS kolNZP 
FROM 
(
    SELECT 
        ks.kod,
        LEFT(ks.kod, 7) AS kodd,
        ks.kodd_rt,
        ks.grup,
        ks.articul,
        ks.mod,
        ks.kod_v,
        IIF(ISNULL(ks.kod_v, 0) = 1, rzv.pach_kod, rzu.pach_kod) AS pachKod,
        IIF(ISNULL(ks.kod_v, 0) = 1, rzv.kol, rzu.kol) AS kolRVAll,
        SUM(ISNULL(nr.kol, 0)) AS kolNaklAll,
        SUM(IIF(ISNULL(n.dost_n, 0) <> 0, nr.kol, 0)) AS kolGI
    FROM #articulKodSelect ks
    LEFT JOIN raskr_zeh_up rzu ON ks.kod = rzu.kod AND ISNULL(ks.kod_v, 0) <> 1
    LEFT JOIN raskr_zeh_vyaz rzv ON ks.kod = rzv.kod AND ISNULL(ks.kod_v, 0) = 1
    LEFT JOIN nakl_ras nr ON IIF(ISNULL(ks.kod_v, 0) = 1, rzv.pach_kod, rzu.pach_kod) = nr.pach_kod
    LEFT JOIN nakl n ON nr.iz = n.iz
    GROUP BY 
        ks.kod,
        LEFT(ks.kod, 7),
        ks.kodd_rt,
        ks.grup,
        ks.articul,
        ks.mod,
        ks.kod_v,
        IIF(ISNULL(ks.kod_v, 0) = 1, rzv.kol, rzu.kol),
        IIF(ISNULL(ks.kod_v, 0) = 1, rzv.pach_kod, rzu.pach_kod)
) s
GROUP BY 
    s.kodd_rt, 
    s.kodd, 
    s.grup, 
    s.articul, 
    s.mod, 
    s.kod_v
ORDER BY 
    s.kodd_rt, 
    s.kodd, 
    s.grup, 
    s.articul, 
    s.mod, 
    s.kod_v;
END

GO

CREATE PROCEDURE dbo.getTbAndImageForRaszReport
    @annId INT
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE @kod CHAR(7), @path NVARCHAR(150), @tb_Id VARCHAR(20);

    -- Получаем kod по annId
    SELECT TOP 1 @kod = vsa.kod
    FROM dbo.View_sp_articul vsa
    WHERE vsa.annId = @annId;

    -- Основная логика поиска файла и tbId
    SELECT TOP 1
        @path = N'h:\proizv\eskiz\' + TRIM(psa.nn) + N'.jpg',
        @tb_Id = psa.tb_Id
    FROM (
        SELECT kodd, MAX(nn) nn, MAX(tb_Id) tb_Id
        FROM plan_sezon_all
        WHERE pict_prn = 1
        GROUP BY kodd
    ) psa
    WHERE psa.kodd = @kod;

    IF ISNULL(@path, N'') = N''
    BEGIN
        SET @path = N'h:\proizv\picture\' + TRIM(@kod) + N'.jpg';
        SET @tb_Id = NULL;
    END

    SELECT @path AS FileEskizPath, @tb_Id AS tb_Id, @kod AS kod;
END
